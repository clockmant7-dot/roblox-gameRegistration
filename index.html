Этот скрипт проверяет код активации. Вставьте его в ServerScriptService

ActivationScript.lua
-- 🔑 Скрипт активации Roblox Studio -- Вставить в: ServerScriptService local activationCode = "XXXX-XXXX-XXXX" local playerName = "Игрок" local function checkActivation(code) if code == activationCode then print("✅ Активация успешна!") print("👤 Разработчик: " .. playerName) print("🔑 Код принят: " .. code) return true else warn("❌ Неверный код активации!") return false end end checkActivation(activationCode)

-- 🖥️ Скрипт создания GUI -- Вставить в: StarterGui → LocalScript local player = game.Players.LocalPlayer -- Создаём ScreenGui local screenGui = Instance.new("ScreenGui") screenGui.Parent = player.PlayerGui -- Создаём рамку local frame = Instance.new("Frame") frame.Size = UDim2.new(0, 300, 0, 100) frame.Position = UDim2.new(0.5, -150, 0, 20) frame.BackgroundColor3 = Color3.fromRGB(30, 30, 50) frame.Parent = screenGui -- Текст приветствия local label = Instance.new("TextLabel") label.Size = UDim2.new(1, 0, 1, 0) label.Text = "👋 Привет, " .. player.Name .. "!" label.TextColor3 = Color3.fromRGB(0, 162, 255) label.TextSize = 22 label.BackgroundTransparency = 1 label.Parent = frame -- Скрыть через 5 секунд task.wait(5) screenGui:Destroy()

Создаёт таблицу очков (Leaderboard). Вставьте в ServerScriptService

LeaderstatsScript.lua
-- 🏆 Скрипт таблицы очков -- Вставить в: ServerScriptService game.Players.PlayerAdded:Connect(function(player) -- Создаём папку leaderstats local leaderstats = Instance.new("Folder") leaderstats.Name = "leaderstats" leaderstats.Parent = player -- Монеты local coins = Instance.new("IntValue") coins.Name = "💰 Монеты" coins.Value = 0 coins.Parent = leaderstats -- Уровень local level = Instance.new("IntValue") level.Name = "⭐ Уровень" level.Value = 1 level.Parent = leaderstats -- Даём монеты каждые 10 сек while true do task.wait(10) coins.Value = coins.Value + 5 end end)

📁 Куда вставить:

Explorer → ServerScriptService → Script

Открывает дверь когда игрок подходит. Вставьте скрипт внутрь Part (двери)

DoorScript.lua
-- 🚪 Скрипт автоматической двери -- Вставить в: Part (дверь) → Script local door = script.Parent local isOpen = false local debounce = false door.Touched:Connect(function(hit) if debounce then return end local player = game.Players:GetPlayerFromCharacter(hit.Parent) if not player then return end debounce = true -- Открыть дверь door.Transparency = 1 door.CanCollide = false print("🚪 Дверь открылась!") -- Закрыть через 3 секунды task.wait(3) door.Transparency = 0 door.CanCollide = true print("🚪 Дверь закрылась!") task.wait(1) debounce = false end)

📁 Куда вставить:

Explorer → Workspace → Part (дверь) → Script

Простой магазин с покупкой скорости за монеты. Вставьте в ServerScriptService

ShopScript.lua
-- 🛒 Скрипт магазина -- Вставить в: ServerScriptService local ReplicatedStorage = game:GetService("ReplicatedStorage") -- Создаём Remote Event для магазина local buyEvent = Instance.new("RemoteEvent") buyEvent.Name = "BuyItem" buyEvent.Parent = ReplicatedStorage -- Товары магазина local shop = { ["Скорость"] = { price = 50, speed = 30 }, ["Турбо"] = { price = 100, speed = 50 } } -- Обработка покупки buyEvent.OnServerEvent:Connect(function(player, item) local stats = player:FindFirstChild("leaderstats") if not stats then return end local coins = stats:FindFirstChild("💰 Монеты") local product = shop[item] if product and coins.Value >= product.price then coins.Value = coins.Value - product.price local char = player.Character if char then char.Humanoid.WalkSpeed = product.speed end print("🛒 Куплено: " .. item) else warn("❌ Недостаточно монет!") end end)

📁 Куда вставить:

Explorer → ServerScriptService → Script
